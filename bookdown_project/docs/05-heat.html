6.1 Environmental Data Preparation

Our first step is to load and transform the hospital data. Temperature data was sourced from the weathercan R package.

#load the data
odhf_valid <- read.csv("./data/odhf_hospitals.csv", row.names = 1)
survey_sf <- odhf_valid %>% st_as_sf(coords=c("long", "lat"), crs=st_crs(4326)) %>% select(-facility_name)
survey_sf <- st_transform(survey_sf, 3978)
stations_list <- stations() 
names(stations_list)

In order to collect the relevant weather stations and time frames, the minimum and maximum survey data collection periods needed to be collected from the ODHF data.

#identify the date range
survey_sf <- survey_sf %>%
  mutate(
    weather_start = make_date(survey_year - 2, survey_month, 1),  #Start date: 2 years before
    weather_end = make_date(survey_year, survey_month, days_in_month(make_date(survey_year, survey_month, 1))))

min_start_year <- min(year(survey_sf$weather_start), na.rm = TRUE)
max_end_year <- max(year(survey_sf$weather_end), na.rm = TRUE)

stations_by_day <- filter(stations_list, 
                                      interval %in% c("hour","day"), 
                                      #setting the min coverage years for data at the stations
                                      #is to increase likelihood that the same hospital location will be assigned                                          to the same weather station
                                      #across survey years 
                                      #this will reduce noise introduced as a result of pulling data for the same                                         location from different sources 
                                      #which may have different data available, different frequencies of                                                 collection, etc. 
                                      start <= min_start_year, 
                                      end >= max_end_year)

Next, we need to get the nearest distance to stations for evaluation of if they are too close or too far away. In the case of this example, no station will be too far away.

#get the stations by day
stations_by_day_sf <- st_as_sf(stations_by_day, coords=c("lon", "lat"), crs = 4326)
stations_by_day_sf <- st_transform(stations_by_day_sf, st_crs(survey_sf))
#get the nearest feature
nearest_stations <- st_nearest_feature(survey_sf, stations_by_day_sf)
#get the nearest distances and ids of the stations for each hospital
survey_sf$nearest_dist <- st_distance(survey_sf, stations_by_day_sf[nearest_stations,], by_element = TRUE)
survey_sf$near_station_id <- stations_by_day_sf[nearest_stations,]$station_id
survey_sf$station_interval <- stations_by_day_sf[nearest_stations,]$interval
survey_sf <- survey_sf %>% rename(station_id=near_station_id)

Here, we get the data from the weathercan package.

#query the package for every station that was identified
hourly_station_ids <- filter(survey_sf, station_interval == "hour") %>% pull(station_id)
daily_station_ids <- filter(survey_sf, station_interval == "day") %>% pull(station_id)
daily_weather <- weather_dl(station_ids = daily_station_ids, start=min(survey_sf$weather_start), end=max(survey_sf$weather_end), interval='day')
hourly_weather <- weather_dl(station_ids = hourly_station_ids, start=min(survey_sf$weather_start), end=max(survey_sf$weather_end), interval='hour')

Then, we group and combine the hourly with the daily so a comprehensive dataset is created.

#summarize from hourly to daily and combine
hourly_as_daily_weather_original <- hourly_weather %>% 
  group_by(station_id, date, lat, lon) %>% 
  summarise(
    min_temp = min(temp, na.rm = FALSE),
    max_temp = max(temp, na.rm = FALSE),
    .groups = "drop"
  )
hourly_as_daily_weather <- subset(hourly_as_daily_weather_original, select = -c(lat, lon))
combined_hourly_daily <- bind_rows(hourly_as_daily_weather, daily_weather) %>%
  select(any_of(intersect(names(hourly_as_daily_weather), names(daily_weather))))
write_parquet(combined_hourly_daily, r"(./data/odhf_weather_stations.parquet)")

6.2 Heat ECCC

Environment and Climate Change Canada has their own criteria for extreme heat events around the country. Using a rough sketch of the criteria map, we can relate the hospital data to the temperature data.

combined_hourly_daily <- read_parquet(r"(./data/odhf_weather_stations.parquet)")
eccc_friendly <- st_read(r"(./data/roughEcccMap_final.gpkg)")
survey_sf_eccc <- st_join(survey_sf, eccc_friendly)

consecutive_days <- 2 #2 days and nights as the criteria.


#identify days of consecutive heat
heat_flagged_eccc <- survey_sf_eccc %>%
  left_join(combined_hourly_daily, by='station_id') %>%  #date, station_id, max_temp, min_temp 
  unique(.,by=c("hospital_id", "survey_month", "survey_year", "date")) %>%  #get unique entries based on survey_id, survey month + year, and date of station data collection
  arrange(station_id, date) %>% #need dates in order chronologically
  group_by(station_id, hospital_id, survey_month, survey_year) %>% #station_ids grouped together
  mutate(
    hot = max_temp >= day_thresh & min_temp >= nite_thres, #if the max_temp is greater than the day threshold AND the minimum temperature is greater than the nighttime threshold
    run_id = rleid(hot) #add unique id for runs
  ) %>% 
  group_by(hospital_id, survey_month, survey_year, station_id, run_id) %>% #group by station id and the run ids (looking for two or more)
  mutate(
    run_len = sum(hot), #sum the number of hot events per run                                    
    heat_event = hot & run_len >= consecutive_days #only classed as a heat event if two or more days       
  ) %>%
  ungroup() %>%
  select(-run_id, -run_len) %>% 
  mutate(date = as.Date(date))

6.2.1 Date Lag

As part of the exposure determination, we are interested in the 24 months prior to each survey date at the hospitals. Hospitals were surveyed at different times, so the date of the survey will be recorded along with each month from the 0th to 24th month prior (survey month, survey month - 1, survey month - 2, etc.).

expand_months <- function(d, hospital_id, survey_month, survey_year, station_id) {
  data.table(
    hospital_id = hospital_id,
    survey_year = survey_year,
    survey_month = survey_month,
    station_id = station_id,
    offset = 0:24,
    date = format(ceiling_date(d %m-% months(0:24), "month") - days(1), "%Y%m")
  )
}

#apply the function
#identifies how many months a given month is offset from the 0th month for that surveys survey date
survey_sf$dates <- ymd(paste(survey_sf$survey_year, survey_sf$survey_month, "01", sep = "-")) + months(1) - days(1)
#FOR EVERY UNIQUE survey AND SURVEY DATE COMBINATION <- this is key
surveys_un <- unique(survey_sf, by = c("survey_id","dates"))
surveys_un <- as.data.table(surveys_un)
survey_expanded <- surveys_un[, expand_months(dates, hospital_id, survey_month, survey_year, station_id), by = .(hospital_id, dates)]
survey_expanded <- survey_expanded %>% select(-3)
survey_expanded_heat_eccc <- survey_expanded %>% rename(year_month = date)

6.2.2 Date Join

Create heat event data for all months before narrowing the range and then join with the expanded days.

#arrange the data chronologically once again and setup columns for joining, add another heat classifier since this is being grouped to months (month has_heat)
heat_monthly_eccc <- heat_flagged_eccc %>%
  st_drop_geometry() %>% #drop the geometry for faster speeds
  arrange(station_id, date) %>%
  mutate(year_month = as.integer(format(date, "%Y%m"))) %>%
  group_by(hospital_id, year_month) %>%
  summarise(
    has_heat = any(heat_event, na.rm = TRUE),                     
    .groups = "drop"
  ) %>%
  mutate(
    year  = as.integer(year_month / 100), #need a year eventually
    month = as.integer(year_month %% 100), #need a month eventually
    year_month = as.character(year_month)
  )
  

#join with the survey data
survey_with_heat_eccc <- survey_expanded_heat_eccc %>% #select(-5,-6) %>% 
  left_join(heat_monthly_eccc, 
            by = c("hospital_id", "year_month")) %>%
  #rename(year_month=dates) %>% 
  mutate(
    heat_flag = as.integer(coalesce(has_heat, FALSE))) #integer based has_heat           

heat_month_zero_daily_eccc <- filter(survey_with_heat_eccc, survey_year==year & survey_month==month) %>% st_drop_geometry()

6.2.3 Matrix

survey_with_heat_eccc has all the interesting information in it about the matrix of heat.

#create the summary matrix of 1's and 0's for the months
heat_summary_matrix_eccc <- survey_with_heat_eccc %>% 
  filter(heat_flag == 1) %>% #only the months with floods in them
  group_by(hospital_id, survey_year, survey_month, offset) %>% #group by the offsets
  slice_head(n=1) %>%
  right_join(survey_expanded_heat_eccc, by=c("hospital_id", "survey_year", "survey_month", "offset")) %>%
  mutate(heat_flag = replace_na(heat_flag, 0)) %>% #make sure there are no NAs
  ungroup() %>%
  as.data.table() %>%
  left_join(survey_sf %>% select(hospital_id, station_id, nearest_dist), by='hospital_id') %>%
  dcast(.,
      hospital_id + survey_month + survey_year + station_id + nearest_dist~ paste0("month_", offset),
      value.var = "heat_flag",
      fun.aggregate = max
    ) %>%
  setcolorder(c("hospital_id", "survey_month", "survey_year", 'station_id', 'nearest_dist', paste0("month_", 0:24))) 

write.csv(heat_summary_matrix_eccc, r"(./data/heat_matrix_eccc.csv)")

combined_hourly_daily <- read_parquet(r"(./data/odhf_weather_stations.parquet)")
temp_threshold <- 30

heat_flagged <- combined_hourly_daily %>%
  arrange(station_id, date) %>% #arrange by station and chronologically
  unique(.,by=c("station_id", "date")) %>% #filter by only unqiue stations and dates, can't have any duplicates

    run_len = sum(hot), #sum the number of hot events per run

  ) %>%
  ungroup() %>%
  select(-run_id, -run_len) %>% 
  mutate(date = as.Date(date))

6.4 Map

The South Okanagan General Hospital in British Columbia experienced a heat event, as defined by Environment and Climate Change Canada, in the 18th month prior to the survey administration month. The Hospital is located in the Southeastern British Columbia zone, where the daytime max temperature threshold is 35C and the nighttime minimum threshold temperature is 18C. The weather station and hospital locations are presented below.

#get hospital 509, heat event in month 18
